    Gallery Mobile Optimization Guide¶

    Overview¶

    The Gallery System is fully optimized for mobile devices, providing a seamless experience for photo management on smartphones and tablets. This guide covers mobile-specific features, optimizations, and best practices.

    Mobile Features¶

    Core Mobile Features¶

    ✅ Camera Capture - Direct photo capture using device camera - Support for front and rear cameras - Multiple photo capture in succession

    ✅ Touch-Optimized Interface - Large touch targets (minimum 44px) - Gesture-friendly interactions - Swipe navigation for photo galleries

    ✅ Responsive Upload - Mobile-optimized batch upload component - Reduced concurrent uploads for mobile networks - Progress indicators optimized for small screens

    ✅ Adaptive Quality - Automatic image compression for mobile - Reduced initial image sizes - Progressive loading with placeholders

    ✅ Offline Capabilities - Basic offline functionality - Queue uploads for when connection returns - Local storage of upload progress

    Touch Interactions¶

    Gesture Support¶

    Touch gestures for enhanced mobile experience:

    // Swipe detection for image navigation
const useSwipeGesture = (onSwipeLeft: () => void, onSwipeRight: () => void) => {
  const [touchStart, setTouchStart] = useState<number | null>(null);
  const [touchEnd, setTouchEnd] = useState<number | null>(null);

  const minSwipeDistance = 50;

  const onTouchStart = (e: TouchEvent) => {
    setTouchEnd(null);
    setTouchStart(e.targetTouches[0].clientX);
  };

  const onTouchMove = (e: TouchEvent) => {
    setTouchEnd(e.targetTouches[0].clientX);
  };

  const onTouchEnd = () => {
    if (!touchStart || !touchEnd) return;

    const distance = touchStart - touchEnd;
    const isLeftSwipe = distance > minSwipeDistance;
    const isRightSwipe = distance < -minSwipeDistance;

    if (isLeftSwipe) onSwipeLeft();
    if (isRightSwipe) onSwipeRight();
  };

  return { onTouchStart, onTouchMove, onTouchEnd };
};

    Pull-to-Refresh¶

    Native-like pull-to-refresh functionality:

    const usePullToRefresh = (onRefresh: () => void) => {
  const [isRefreshing, setIsRefreshing] = useState(false);
  const [pullDistance, setPullDistance] = useState(0);

  useEffect(() => {
    let startY = 0;
    let currentY = 0;

    const handleTouchStart = (e: TouchEvent) => {
      startY = e.touches[0].clientY;
    };

    const handleTouchMove = (e: TouchEvent) => {
      currentY = e.touches[0].clientY;
      const distance = currentY - startY;

      if (distance > 0 && window.scrollY === 0) {
        setPullDistance(Math.min(distance, 100));
        e.preventDefault();
      }
    };

    const handleTouchEnd = () => {
      if (pullDistance > 60) {
        setIsRefreshing(true);
        onRefresh();
      }
      setPullDistance(0);
    };

    // Add event listeners
    document.addEventListener('touchstart', handleTouchStart);
    document.addEventListener('touchmove', handleTouchMove, { passive: false });
    document.addEventListener('touchend', handleTouchEnd);

    return () => {
      document.removeEventListener('touchstart', handleTouchStart);
      document.removeEventListener('touchmove', handleTouchMove);
      document.removeEventListener('touchend', handleTouchEnd);
    };
  }, [pullDistance, onRefresh]);

  return { isRefreshing, pullDistance, setIsRefreshing };
};

    Lazy Loading Strategy¶

    Efficient lazy loading for mobile:

    const useLazyLoading = () => {
  const [isIntersecting, setIsIntersecting] = useState(false);
  const ref = useRef<HTMLElement>(null);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsIntersecting(true);
          observer.disconnect();
        }
      },
      {
        rootMargin: '100px', // Start loading 100px before visible
        threshold: 0.1
      }
    );

    if (ref.current) {
      observer.observe(ref.current);
    }

    return () => observer.disconnect();
  }, []);

  return { ref, isIntersecting };
};

    Network Considerations¶

    Adaptive Loading¶

    Network-aware loading strategies:

    const useNetworkAware = () => {
  const [networkInfo, setNetworkInfo] = useState({
    effectiveType: '4g',
    downlink: 10,
    saveData: false
  });

  useEffect(() => {
    const connection = (navigator as any).connection;
    if (connection) {
      setNetworkInfo({
        effectiveType: connection.effectiveType,
        downlink: connection.downlink,
        saveData: connection.saveData
      });

      const updateConnection = () => {
        setNetworkInfo({
          effectiveType: connection.effectiveType,
          downlink: connection.downlink,
          saveData: connection.saveData
        });
      };

      connection.addEventListener('change', updateConnection);
      return () => connection.removeEventListener('change', updateConnection);
    }
  }, []);

  // Adjust quality based on network
  const getImageQuality = () => {
    if (networkInfo.saveData || networkInfo.effectiveType === 'slow-2g') {
      return 'thumbnail';
    }
    if (networkInfo.effectiveType === '2g' || networkInfo.effectiveType === '3g') {
      return 'small';
    }
    return 'medium';
  };

  return { networkInfo, getImageQuality };
};

    Offline Support¶

    Basic offline functionality:

    const useOfflineQueue = () => {
  const [isOnline, setIsOnline] = useState(navigator.onLine);
  const [uploadQueue, setUploadQueue] = useState<PhotoUploadInput[]>([]);

  useEffect(() => {
    const handleOnline = () => setIsOnline(true);
    const handleOffline = () => setIsOnline(false);

    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);

    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
    };
  }, []);

  const queueUpload = (input: PhotoUploadInput) => {
    if (isOnline) {
      uploadPhoto(input);
    } else {
      setUploadQueue(prev => [...prev, input]);
      showOfflineMessage();
    }
  };

  const processQueue = async () => {
    if (isOnline && uploadQueue.length > 0) {
      for (const input of uploadQueue) {
        try {
          await uploadPhoto(input);
        } catch (error) {
          console.error('Failed to upload queued photo:', error);
        }
      }
      setUploadQueue([]);
    }
  };

  useEffect(() => {
    if (isOnline) {
      processQueue();
    }
  }, [isOnline]);

  return { isOnline, queueUpload, uploadQueue };
};

    Performance Issues¶

    // Monitor performance
const monitorPerformance = () => {
  // Memory usage
  if ('memory' in performance) {
    console.log('Memory:', (performance as any).memory);
  }

  // Frame rate
  let lastTime = performance.now();
  let frameCount = 0;

  const measureFPS = () => {
    frameCount++;
    const currentTime = performance.now();

    if (currentTime - lastTime >= 1000) {
      console.log('FPS:', frameCount);
      frameCount = 0;
      lastTime = currentTime;
    }

    requestAnimationFrame(measureFPS);
  };

  measureFPS();
};
